## 4. Lightweight License Key Verification Blueprint for Swift (`LemonSqueezyLicenseManager`)

下方提供零第三方依赖、完全兼容 **Swift 6.0 Strict Concurrency** 的 `LemonSqueezyLicenseManager` 实现，直接对接 Lemon Squeezy 官方公开端点：
- `POST https://api.lemonsqueezy.com/v1/licenses/activate`
- `POST https://api.lemonsqueezy.com/v1/licenses/validate`

> **安全优势**：Lemon Squeezy 的 `/v1/licenses/*` 端点专为客户端软件设计，**不需要在客户端硬编码任何商户 API Key Secret**，仅凭用户输入的 `license_key` 和设备 `instance_id` 即可完成激活、设备数限制校验与 Variant 层级识别（自动区分 `$69 Indie` 与 `$149 Agency White-Label`）。

### Ready-to-Integrate Swift Snippet (Drop into `Sources/GEORadarPro/main.swift`)

```swift
import Foundation
import SwiftUI

// MARK: - Lemon Squeezy License Models & Manager (Swift 6 Sendable)

struct LemonLicenseResponse: Codable, Sendable {
    let activated: Bool?
    let valid: Bool?
    let error: String?
    let licenseKey: LicenseKeyInfo?
    let instance: InstanceInfo?
    let meta: MetaInfo?

    enum CodingKeys: String, CodingKey {
        case activated, valid, error, instance, meta
        case licenseKey = "license_key"
    }

    struct LicenseKeyInfo: Codable, Sendable {
        let id: Int
        let status: String          // "active", "inactive", "expired", "disabled"
        let key: String
        let activationLimit: Int?
        let activationUsage: Int?

        enum CodingKeys: String, CodingKey {
            case id, status, key
            case activationLimit = "activation_limit"
            case activationUsage = "activation_usage"
        }
    }

    struct InstanceInfo: Codable, Sendable {
        let id: String
        let name: String
    }

    struct MetaInfo: Codable, Sendable {
        let storeId: Int?
        let productId: Int?
        let variantId: Int?
        let variantName: String?
        let customerEmail: String?

        enum CodingKeys: String, CodingKey {
            case storeId = "store_id"
            case productId = "product_id"
            case variantId = "variant_id"
            case variantName = "variant_name"
            case customerEmail = "customer_email"
        }
    }
}

enum LicenseTier: String, Sendable {
    case unlicensed = "Trial / Unlicensed"
    case indiePro = "Indie / In-House Pro ($69)"
    case agencyWhiteLabel = "Agency White-Label ($149)"
}

@MainActor
final class LemonSqueezyLicenseManager: ObservableObject {
    static let shared = LemonSqueezyLicenseManager()

    @Published var licenseKey: String = UserDefaults.standard.string(forKey: "geo_ls_license_key") ?? ""
    @Published var instanceId: String = UserDefaults.standard.string(forKey: "geo_ls_instance_id") ?? ""
    @Published var tier: LicenseTier = .unlicensed
    @Published var statusMessage: String = "Enter your Lemon Squeezy license key to unlock."
    @Published var isVerifying: Bool = false

    private let activateURL = URL(string: "https://api.lemonsqueezy.com/v1/licenses/activate")!
    private let validateURL = URL(string: "https://api.lemonsqueezy.com/v1/licenses/validate")!

    var isAgencyUnlocked: Bool {
        tier == .agencyWhiteLabel
    }

    var isProUnlocked: Bool {
        tier == .indiePro || tier == .agencyWhiteLabel
    }

    /// Activate a new license key on this Mac (`/v1/licenses/activate`)
    func activateLicense(key: String) async {
        let trimmed = key.trimmingCharacters(in: .whitespacesAndNewlines)
        guard !trimmed.isEmpty else {
            statusMessage = "Please enter a valid license key."
            return
        }

        isVerifying = true
        defer { isVerifying = false }

        let hostName = Host.current().localizedName ?? "macOS-Workstation"
        let bodyParams = [
            "license_key": trimmed,
            "instance_name": hostName
        ]

        do {
            let resp = try await postForm(url: activateURL, params: bodyParams)
            if (resp.activated == true || resp.valid == true),
               let lk = resp.licenseKey, lk.status == "active" {
                let newInstanceId = resp.instance?.id ?? instanceId
                persist(key: trimmed, instanceId: newInstanceId, meta: resp.meta)
                statusMessage = "Activated (\(tier.rawValue))"
            } else {
                statusMessage = resp.error ?? "Activation failed. Check key or activation limit."
            }
        } catch {
            statusMessage = "Network error during activation: \(error.localizedDescription)"
        }
    }

    /// Validate an existing stored license key & instance on launch (`/v1/licenses/validate`)
    func validateStoredLicense() async {
        guard !licenseKey.isEmpty else { return }
        isVerifying = true
        defer { isVerifying = false }

        var params = ["license_key": licenseKey]
        if !instanceId.isEmpty {
            params["instance_id"] = instanceId
        }

        do {
            let resp = try await postForm(url: validateURL, params: params)
            if resp.valid == true, let lk = resp.licenseKey, lk.status == "active" {
                persist(key: licenseKey, instanceId: instanceId, meta: resp.meta)
                statusMessage = "Verified (\(tier.rawValue))"
            } else {
                tier = .unlicensed
                statusMessage = resp.error ?? "License is no longer active."
            }
        } catch {
            // Offline grace fallback: keep cached tier if network is temporarily unreachable
            if let cachedTier = UserDefaults.standard.string(forKey: "geo_ls_cached_tier"),
               let restored = LicenseTier(rawValue: cachedTier) {
                tier = restored
                statusMessage = "Offline Mode (\(restored.rawValue))"
            }
        }
    }

    private func persist(key: String, instanceId: String, meta: LemonLicenseResponse.MetaInfo?) {
        self.licenseKey = key
        self.instanceId = instanceId
        UserDefaults.standard.set(key, forKey: "geo_ls_license_key")
        UserDefaults.standard.set(instanceId, forKey: "geo_ls_instance_id")

        let variantName = (meta?.variantName ?? "").lowercased()
        if variantName.contains("agency") || variantName.contains("white-label") {
            self.tier = .agencyWhiteLabel
        } else {
            self.tier = .indiePro
        }
        UserDefaults.standard.set(self.tier.rawValue, forKey: "geo_ls_cached_tier")
    }

    private func postForm(url: URL, params: [String: String]) async throws -> LemonLicenseResponse {
        var req = URLRequest(url: url)
        req.httpMethod = "POST"
        req.setValue("application/x-www-form-urlencoded", forHTTPHeaderField: "Content-Type")
        req.setValue("application/json", forHTTPHeaderField: "Accept")
        req.timeoutInterval = 10.0

        let formBody = params.map { k, v in
            let ek = k.addingPercentEncoding(withAllowedCharacters: .urlQueryAllowed) ?? k
            let ev = v.addingPercentEncoding(withAllowedCharacters: .urlQueryAllowed) ?? v
            return "\(ek)=\(ev)"
        }.joined(separator: "&")
        req.httpBody = formBody.data(using: .utf8)

        let (data, _) = try await URLSession.shared.data(for: req)
        return try JSONDecoder().decode(LemonLicenseResponse.self, from: data)
    }
}
```

### 集成进入 `main.swift` 的 3 步接线指南

1. **启动时静默校验（Offline-Grace Friendly）**：
   在主视图 `ContentView` 的 `.onAppear`（或 `.task`）中调用：
   ```swift
   .task {
       await LemonSqueezyLicenseManager.shared.validateStoredLicense()
   }
   ```
2. **区分 `$69 Indie` 与 `$149 Agency` 功能门控（Feature Gating）**：
   - 当 `LemonSqueezyLicenseManager.shared.isAgencyUnlocked == false` 时，将侧边栏的 `Agency Name` / `Client Name` 输入框锁定为默认的水印值 `"Generated by GEO Radar Pro (Indie Edition)"`，并在点击 `Open Agency HTML Report` 时展示升级至 `$149 Agency White-Label Edition` 的购买引导链接。
   - 当检测到 Lemon Squeezy 返回的 `meta.variant_name` 包含 `"Agency"` 时，自动解锁自定义白标署名与无水印 HTML/PDF 导出。
3. **离线容错保护（Offline Grace Fallback）**：
   如上方代码所示，当用户在飞机上或无外网环境打开软件时，`validateStoredLicense()` 的 `catch` 分支会自动读取本地缓存的 `geo_ls_cached_tier`，确保付费用户在离线使用无 Key 先验审计或查看本地 SQLite 历史记录时永远不会被误锁。
